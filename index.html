// ================= MONITORING CENTER LOGIC ================= function openMonitoringCenter() { document.getElementById('dashboard-container').style.display = 'none'; document.getElementById('monitoring-overlay').style.display = 'block'; const tbody = document.getElementById('mon-tbody'); tbody.innerHTML = ""; const allProjects = {...projectData}; for (const cId in customProjects) { if(!allProjects[cId]) allProjects[cId] = customProjects[cId].data; } for (const pId in allProjects) { if(deletedProjects.includes(pId)) continue; let pName = allProjects[pId].title.replace("WEEKLY PROJECT UPDATES ", ""); let responsible = allProjects[pId].responsible || "TBD"; let lastUpdated = "Default Data"; let redFlagsHTML = ""; let totalFlags = 0; // Scan all saved weeks/months for this project in stateDB to find the active red flags card let foundCardHTML = null; for (const key in stateDB) { if (key.startsWith(pId + "_")) { foundCardHTML = stateDB[key].html; lastUpdated = key.replace(pId + "_", "").replace("_", " "); } } if (foundCardHTML) { const tempDiv = document.createElement('div'); tempDiv.innerHTML = foundCardHTML; // Extract directly from the KEY GAPS & RED FLAGS card container const rfCard = tempDiv.querySelector('#card-red-flags') || tempDiv.querySelector('.light-card'); if (rfCard) { const items = rfCard.querySelectorAll('.red-flag-item div[contenteditable="true"]'); totalFlags = items.length; items.forEach(item => { if(item.innerText.trim() !== "") { redFlagsHTML += `
  • ${item.innerText}
    • `; } }); } } // Fallback to initial data array if no saved state exists yet if (totalFlags === 0 && allProjects[pId].redFlags) { const flags = allProjects[pId].redFlags; totalFlags = flags.length; flags.forEach(f => { redFlagsHTML += `
  • ${f}
    • `; }); } if(totalFlags === 0 || redFlagsHTML === "") { redFlagsHTML = "
  • No red flags reported.
    • "; } tbody.innerHTML += ` ${pName} ${responsible} ${lastUpdated} ${totalFlags}
      ${redFlagsHTML}
    `; } }
