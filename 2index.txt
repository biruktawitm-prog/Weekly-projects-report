    <script>
        // Simple local storage persistence for edits
        function saveData() {
            const elements = document.querySelectorAll('[contenteditable="true"]');
            const data = {};
            elements.forEach((el, idx) => {
                const id = el.id || 'editable_' + idx;
                data[id] = el.innerHTML;
            });
            localStorage.setItem('unops_dashboard_data', JSON.stringify(data));
            alert('Dashboard changes saved successfully!');
        }

        function loadData() {
            const saved = localStorage.getItem('unops_dashboard_data');
            if (saved) {
                const data = JSON.parse(saved);
                Object.keys(data).forEach(id => {
                    const el = document.getElementById(id);
                    if (el) el.innerHTML = data[id];
                });
            }
        }

        window.onload = loadData;
    </script>
